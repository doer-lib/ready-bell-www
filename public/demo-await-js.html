An async API as an async function

A slow job behind two REST requests reads best as one line of code: const result = await callAsyncApi("Alice", 7). Here is how to get there with polling, and how Ready Bell makes the same function faster and lighter.

1. Wrap the REST calls in an async function

The hello-async API at demo.ready-bell.com is asynchronous: POST /hello-async/{uuid} starts a job and returns at once, and GET /hello-async/{uuid} reports its status until it is READY or FAILED. In Node.js, all of that fits inside an async function. It submits the job, then loops: sleep, check, and return the job once it has finished. The caller just awaits the result, and never sees the polling.

call-async-api-poll-only.js View raw 
// Usage: node call-async-api-poll-only.js [name] [dbg-delay-sec]

import { setTimeout as sleep } from "timers/promises";

const API = "https://demo.ready-bell.com/hello-async";
const POLL_INTERVAL_MS = 3000;
const TIMEOUT_MS = 60000;

const [name = "Alice", delaySec = "4"] = process.argv.slice(2);
const started = performance.now();
const result = await callAsyncApi(name, Number(delaySec));
console.log(result);
console.log(`callAsyncApi took ${((performance.now() - started) / 1000).toFixed(3)} s`);

async function callAsyncApi(name, delaySec) {
  const uuid = crypto.randomUUID();
  const deadline = Date.now() + TIMEOUT_MS;

  await submitAsyncJob(uuid, name, delaySec);
  while (Date.now() < deadline) {
    await sleep(POLL_INTERVAL_MS);
    const job = await fetchFinishedJob(uuid).catch(err => console.log(err.message));
    if (job) return job;
  }
  throw new Error(`Timed out after ${TIMEOUT_MS} ms`);
}

function submitAsyncJob(uuid, name, debugDelay) {
  return fetch(`${API}/${uuid}`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ input: { name, "dbg-delay-sec": debugDelay } }),
  }).then(res => {
    if (!res.ok) throw new Error(`POST failed: HTTP ${res.status}`);
  });
}

function fetchFinishedJob(uuid) {
  return fetch(`${API}/${uuid}`).then(res =>
    res.ok ? res.json() : Promise.reject(new Error(`GET failed: HTTP ${res.status}`))
  ).then(job =>
    job.status === "IN_PROGRESS" ? Promise.reject(new Error("Job IN_PROGRESS")) : job
  );
}

The poll interval is a trade-off. Poll rarely and the result waits for the next check; poll often and most requests only learn that the job is still in progress. With a 3-second interval, a 4-second job takes about 6.7 s and two GET requests.

2. Ready Bell as a replacement for sleep()

Ready Bell provides readyBell.wait(uuid, ms) as a replacement for sleep(). It sends Listen <uuid> over UDP and resolves when either the timeout expires or a matching Ready <uuid> notification arrives, whichever comes first. Notifications received while no wait is active — for example, during a GET request — are stored in a backlog, so the next wait resolves immediately.

The implementation is straightforward:

ready-bell.js View raw 
import dgram from "dgram";

const BACKLOG_TTL_MS = 60000;

export class ReadyBell {
  waiters = new Map();
  backlog = new Map();
  socket = dgram.createSocket("udp4");

  constructor() {
    this.socket.on("message", msg => this.onMessage(msg.toString().trim()));
    this.socket.unref();
  }

  sendListen(uuid, seconds) {
    this.evictStale();
    this.socket.send(`Listen ${uuid} ${seconds}`, 3137, "ready-bell.com");
  }

  sendNotify(uuid) {
    this.socket.send(`Notify ${uuid}`, 3137, "ready-bell.com");
  }

  wait(uuid, ms) {
    if (this.backlog.delete(uuid)) return Promise.resolve();
    this.sendListen(uuid, 2 * ms / 1000);
    return new Promise(resolve => {
      const wakeUp = () => {
        clearTimeout(timer);
        this.waiters.delete(uuid);
        resolve();
      };
      const timer = setTimeout(wakeUp, ms);
      this.waiters.set(uuid, wakeUp);
    });
  }

  onMessage(text) {
    console.log(text);
    const [command, uuid] = text.split(" ");
    if (command !== "Ready") return;
    this.evictStale();
    const wakeUp = this.waiters.get(uuid);
    if (wakeUp) wakeUp();
    else this.backlog.set(uuid, Date.now() + BACKLOG_TTL_MS);
  }

  evictStale() {
    for (const [uuid, expires] of this.backlog) {
      if (expires < Date.now()) this.backlog.delete(uuid);
    }
  }
}

3. The same function, woken up by Ready Bell

callAsyncApi keeps its shape. The highlighted lines are the only ones that use ready-bell.js: subscribe before the POST, so an early Ready isn't missed, and wait instead of sleep. Polling stays as the fallback, because UDP gives no delivery guarantee, so the interval can grow to 15 seconds. The same 4-second job now takes about 4.4 s and one GET.

  • Put call-async-api.js and ready-bell.js in one directory
  • Run node call-async-api-poll-only.js Alice 4, then node call-async-api.js Alice 4, and compare the times (Node.js 22 or newer)

// Usage: node call-async-api.js [name] [dbg-delay-sec]

import { ReadyBell } from "./ready-bell.js";

const API = "https://demo.ready-bell.com/hello-async";
const POLL_INTERVAL_MS = 15000;
const TIMEOUT_MS = 60000;

const readyBell = new ReadyBell();
const [name = "Alice", delaySec = "4"] = process.argv.slice(2);
const started = performance.now();
const result = await callAsyncApi(name, Number(delaySec));
console.log(result);
console.log(`callAsyncApi took ${((performance.now() - started) / 1000).toFixed(3)} s`);

async function callAsyncApi(name, delaySec) {
  const uuid = crypto.randomUUID();
  const deadline = Date.now() + TIMEOUT_MS;

  readyBell.sendListen(uuid, 30);
  await submitAsyncJob(uuid, name, delaySec);
  while (Date.now() < deadline) {
    await readyBell.wait(uuid, POLL_INTERVAL_MS);
    const job = await fetchFinishedJob(uuid).catch(err => console.log(err.message));
    if (job) return job;
  }
  throw new Error(`Timed out after ${TIMEOUT_MS} ms`);
}

}
